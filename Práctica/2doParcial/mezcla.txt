ACCION ejer ES
    AMBIENTE
        empleados = REGISTRO
            dni: entero
            ape: AN(30)
        FR

        archA, archB, archS: ARCHIVO DE empleados ORDENADO POR dni
        regA, regB, regS: empleados

        PROCEDIMIENTO leerA() ES
            LEER(archA, regA)
            SI FDA(archA) ENTONCES
                regA.dni:= HV
            FS
        FP

        PROCEDIMIENTO leerB() ES
            LEER(archB, regB)
            SI FDA(archB) ENTONCES
                regB.dni:= HV
            FS
        FP

        PROCEDIMIENTO inic() ES
            ABRIR E/(archA)
            ABRIR E/(archB)
            ABRIR /S(archS)
            leerA()
            leerB()
        FP

        PROCEDIMIENTO fina() ES
            CERRAR(archA)
            CERRAR(archB)
            CERRAR(archS)
        FP
        
    PROCESO
        inic()
        MIENTRAS (regA.clave <> HV) o (regB.clave <> HV) HACER
            SI (regA.clave < regB.clave) ENTONCES
                regS:= regA
                GRABAR(archS, regS)
                leerA()
            SINO
                SI (regA.clave > regB.clave) ENTONCES
                    regS:= regB
                    GRABAR(archS, regS)
                    leerB()
                FS
            FS
        FM
        fina()
FIN ACCION

ACCION ejer3 ES
    AMBIENTE
        alumnos = REGISTRO
            lu: entero
            ape: AN(30)
            añoCur: 1..5
        FR

        archA, archB, archS: ARCHIVO DE alumnos ORDENADO POR lu
        regA, regB, regS: alumnos

        cantI: entero

        PROCEDIMIENTO leerA() ES
            LEER(archA, regA)
            SI FDA(archA) ENTONCES
                regA.lu:= HV
            FS
        FP

        PROCEDIMIENTO leerB() ES
            LEER(archB, regB)
            SI FDA(archB) ENTONCES
                regB.lu:= HV
            FS
        FP

        PROCEDIMIENTO inic() ES
            ABRIR E/(archA)
            ABRIR E/(archB)
            ABRIR /S(archS)
            leerA()
            leerB()
            cantI:= 0
        FP

        PROCEDIMIENTO fina() ES
            CERRAR(archA)
            CERRAR(archB)
            CERRAR(archS)
        FP
        
    PROCESO
        inic()
        MIENTRAS (regA.lu <> HV) o (regB.lu <> HV) HACER
            SI (regA.lu < regB.lu) ENTONCES
                SI (regA.añoCur = 1) ENTONCES
                    cantI:= cantI + 1
                FS
                regS:= regA
                GRABAR(archS, regS)
                leerA()
            SINO
                SI (regA.lu > regB.lu) ENTONCES
                    SI (regB.añoCur = 1) ENTONCES
                        cantI:= cantI + 1
                    FS
                    regS:= regB
                    GRABAR(archS, regS)
                    leerB()
                FS
            FS
        FM
        ESC("La cantidad de ingresantes fue ", cantI)
        fina()
FIN ACCION

ACCION ejer6 ES
    AMBIENTE
        vip = REGISTRO
            nroCli: entero
            ape: AN(30)
            puntoA: entero
        FR

        deudores = REGISTRO
            nroCli: entero
            ape: AN(30)
            saldoD: real
        FR

        padron = REGISTRO
            nroCli: entero
            ape: AN(30)
            puntos: entero
            deuda: real
        FR

        archA: ARCHIVO DE vip ORDENADO POR nroCli
        regA: vip

        archB: ARCHIVO DE deudores ORDENADO POR nroCli
        regB: deudores

        archS: ARCHIVO DE padron ORDENADO POR nroCli
        regS: padron

        PROCEDIMIENTO leerA() ES
            LEER(archA, regA)
            SI FDA(archA) ENTONCES
                regA.nroCli:= HV
            FS
        FP

        PROCEDIMIENTO leerB() ES
            LEER(archB, regB)
            SI FDA(archB) ENTONCES
                regB.nroCli:= HV
            FS
        FP

        PROCEDIMIENTO inic() ES
            ABRIR E/(archA)
            ABRIR E/(archB)
            ABRIR /S(archS)
            leerA()
            leerB()
        FP

        PROCEDIMIENTO fina() ES
            CERRAR(archA)
            CERRAR(archB)
            CERRAR(archS)
        FP
        
    PROCESO
        inic()
        MIENTRAS (regA.nroCli <> HV) o (regB.nroCli <> HV) HACER
            SI (regA.nroCli < regB.nroCli) ENTONCES

                regS.puntos:= regA.puntoA

                    regS.ape:= regB.ape
                    regS.deuda:= regB.saldoD

                    leerB()
                FS
            FS

        dias: ARREGLO DE [1..7] DE real

        prom:= 0.0

        FP
